#include <U8g2lib.h>               // U8G2 OLED library
#include <Wire.h>                  // I2C library
#include <MAX30100_PulseOximeter.h> // MAX30100 helper wrapper

// --- Shared Hardware I2C Pins ---
#define I2C_SDA_PIN 21
#define I2C_SCL_PIN 22

// --- OLED Display Setup (Hardware I2C sharing the Wire bus) ---
U8G2_SSD1306_128X64_NONAME_F_HW_I2C u8g2(U8G2_R0, U8X8_PIN_NONE);

// --- Pulse Oximeter Setup ---
PulseOximeter pox;
uint32_t lastReportTime = 0;
const unsigned long reportingPeriodMs = 100;

// Waveform graph buffer
const int waveWidth = 128;
int waveBuffer[waveWidth];

// Heartbeat indicator variables
unsigned long lastBeatTime = 0;
bool showHeart = false;

void onBeatDetected() {
  lastBeatTime = millis();
  showHeart = true;
}

void setup() {
  Serial.begin(115200);
  delay(500);

  // Initialize unified Hardware I2C bus on pins 21 & 22
  Wire.begin(I2C_SDA_PIN, I2C_SCL_PIN);
  Wire.setClock(100000); // Standard 100kHz I2C speed for stability

  // Initialize OLED
  u8g2.begin();
  
  u8g2.clearBuffer();
  u8g2.setFont(u8g2_font_ncenB08_tr);
  u8g2.drawStr(10, 35, "Initializing...");
  u8g2.sendBuffer();
  delay(1000);

  // Initialize MAX30100 Pulse Oximeter
  if (!pox.begin()) {
    u8g2.clearBuffer();
    u8g2.drawStr(5, 35, "Sensor Failed!");
    u8g2.sendBuffer();
    for (;;);
  }

  pox.setIRLedCurrent(MAX30100_LED_CURR_7_6MA);
  pox.setOnBeatDetectedCallback(onBeatDetected);

  // Initialize waveform baseline
  for (int i = 0; i < waveWidth; i++) {
    waveBuffer[i] = 58;
  }
}

// Custom heart icon drawing function (filled or outline)
void drawHeartIcon(int x, int y, bool filled) {
  if (filled) {
    u8g2.drawBox(x + 1, y, 2, 2);
    u8g2.drawBox(x + 5, y, 2, 2);
    u8g2.drawBox(x, y + 1, 9, 3);
    u8g2.drawBox(x + 1, y + 4, 7, 2);
    u8g2.drawBox(x + 2, y + 6, 5, 1);
    u8g2.drawBox(x + 3, y + 7, 3, 1);
  } else {
    u8g2.drawPixel(x + 1, y); u8g2.drawPixel(x + 2, y);
    u8g2.drawPixel(x + 5, y); u8g2.drawPixel(x + 6, y);
    u8g2.drawPixel(x, y + 1); u8g2.drawPixel(x + 3, y + 1); u8g2.drawPixel(x + 7, y + 1);
    u8g2.drawPixel(x, y + 2); u8g2.drawPixel(x + 7, y + 2);
    u8g2.drawPixel(x + 1, y + 3); u8g2.drawPixel(x + 6, y + 3);
    u8g2.drawPixel(x + 2, y + 4); u8g2.drawPixel(x + 5, y + 4);
    u8g2.drawPixel(x + 3, y + 5); u8g2.drawPixel(x + 4, y + 5);
  }
}

void loop() {
  pox.update();

  float hr = pox.getHeartRate();
  float spo2 = pox.getSpO2();

  // Update waveform sample periodically
  unsigned long currentMillis = millis();
  if (currentMillis - lastReportTime >= reportingPeriodMs) {
    lastReportTime = currentMillis;
    
    int rawVal = (hr > 30) ? map((int)hr, 40, 160, 62, 50) : 58;
    
    for (int i = 0; i < waveWidth - 1; i++) {
      waveBuffer[i] = waveBuffer[i + 1];
    }
    waveBuffer[waveWidth - 1] = rawVal;
  }

  // Turn off heart flash after 150ms
  if (showHeart && (millis() - lastBeatTime > 150)) {
    showHeart = false;
  }

  // --- Render Display ---
  u8g2.clearBuffer();

  // 1. Header Title Bar
  u8g2.setFont(u8g2_font_6x10_tf);
  u8g2.drawStr(18, 9, "PULSE OXIMETER");
  u8g2.drawHLine(0, 12, 128);

  // 2. Heart Rate Block
  u8g2.drawStr(4, 25, "HR:");
  u8g2.setFont(u8g2_font_fub14_tf);
  char hrBuffer[8];
  if (hr > 30) {
    sprintf(hrBuffer, "%3.0f", hr);
  } else {
    sprintf(hrBuffer, "--");
  }
  u8g2.drawStr(28, 28, hrBuffer);
  u8g2.setFont(u8g2_font_6x10_tf);
  u8g2.drawStr(72, 25, "BPM");

  // 3. SpO2 Block
  u8g2.drawStr(4, 42, "SpO2:");
  u8g2.setFont(u8g2_font_fub14_tf);
  char spo2Buffer[8];
  if (spo2 > 50) {
    sprintf(spo2Buffer, "%3.0f", spo2);
  } else {
    sprintf(spo2Buffer, "--");
  }
  u8g2.drawStr(44, 44, spo2Buffer);
  u8g2.setFont(u8g2_font_6x10_tf);
  u8g2.drawStr(88, 42, "%");

  // 4. Beating Heart Icon on the right side
  drawHeartIcon(108, 22, showHeart);

  // 5. Live Graph (Bottom line divider removed)
  for (int x = 0; x < waveWidth; x++) {
    u8g2.drawPixel(x, waveBuffer[x]);
  }

  u8g2.sendBuffer();
}
